```python
"""TaskMediator seam — one adapter per TaskKind.

The mediator's job: turn a raw NormalizedEnvelope from a RAG tool into the
final string shown to the user. Behaviour varies by TaskKind (summary vs QA
need different prompts, refusal policies, citation handling).

Adding a new TaskKind = add a new adapter + register it. Dispatcher selects
via TaskKind.mediator_id; no caller knows which adapter ran.
"""

from __future__ import annotations

from typing import Protocol

from backend.tool_arena.normalizer import NormalizedEnvelope


class TaskMediator(Protocol):
    """Mediator contract. Async because LLM calls are async."""

    mediator_id: str

    async def mediate(
        self,
        envelope: NormalizedEnvelope,
        task: str,
        goal: str,
        llm_id: str,
    ) -> str:
        ...
```

`backend/tool_arena/task_mediator/registry.py`:

```python
from __future__ import annotations

from backend.tool_arena.task_mediator.base import TaskMediator

_REGISTRY: dict[str, TaskMediator] = {}


def register(mediator: TaskMediator) -> None:
    if mediator.mediator_id in _REGISTRY:
        raise RuntimeError(f"TaskMediator already registered: {mediator.mediator_id}")
    _REGISTRY[mediator.mediator_id] = mediator


def get(mediator_id: str) -> TaskMediator:
    if mediator_id not in _REGISTRY:
        raise KeyError(f"No TaskMediator registered for id={mediator_id}. Known: {sorted(_REGISTRY)}")
    return _REGISTRY[mediator_id]
```
