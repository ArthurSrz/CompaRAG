class CompareRequest(BaseModel):
    task: str = ""           # ignored when haystack == "benchmark"
    goal: str = ""           # ignored when haystack == "benchmark"
    document_content: str = ""
    task_type: Literal["summary", "qa"] | None = None   # drop "extraction" — gone per de_e0442485
    haystack: Literal["benchmark", "sandbox"] = "sandbox"
    evaluation_query_id: str | None = None

    @model_validator(mode="after")
    def _validate_mode(self) -> "CompareRequest":
        if self.haystack == "benchmark":
            if not self.evaluation_query_id:
                raise ValueError("benchmark mode requires evaluation_query_id")
            # task / goal / document_content come from the eval entry — caller may leave them blank
        else:  # sandbox
            if not self.document_content.strip():
                raise ValueError("sandbox mode requires non-empty document_content")
            if self.evaluation_query_id:
                raise ValueError("evaluation_query_id is only valid in benchmark mode")
            if not self.task.strip():
                raise ValueError("sandbox mode requires task")
        return self

```python
"""Server-side evaluation query loader.

Reads the same corpus/evaluation/queries.yaml that rag-pill reads. The two
loaders MUST stay in sync via a shared corpus version hash check at startup.
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import cached_property
from pathlib import Path

import yaml


@dataclass(frozen=True)
class EvaluationQueryMeta:
    id: str
    query_text: str
    goal_text: str


class EvaluationCatalog:
    def __init__(self, queries_path: Path) -> None:
        self._path = queries_path

    @cached_property
    def _by_id(self) -> dict[str, EvaluationQueryMeta]:
        if not self._path.exists():
            return {}
        data = yaml.safe_load(self._path.read_text())
        return {
            entry["id"]: EvaluationQueryMeta(
                id=entry["id"],
                query_text=entry["query_text"],
                goal_text=entry.get("goal_text", ""),
            )
            for entry in data
        }

    def get(self, query_id: str) -> EvaluationQueryMeta | None:
        return self._by_id.get(query_id)

    def list_ids(self) -> list[str]:
        return list(self._by_id.keys())
```
