<objective>
Lift the duplicated `CORPUS_DIR` constant and per-engine corpus-loading into a single `HaystackCorpus` Protocol with two adapters. This is the foundational seam everything else in Phase 13 builds on: it owns "what is the haystack," "where are the source documents," and "what evaluation queries (if any) come with this haystack."

Two adapters earn the seam:
- `FixedCorpus(root=...)` — reads `mcp_servers/corpus/*.md`. `has_ground_truth=True`. Replaces the duplicated `CORPUS_DIR.glob("*.md")` blocks in all 5 engines.
- `EphemeralCorpus(documents=[...])` — wraps user-uploaded `document_content` as a single-doc corpus. `has_ground_truth=False`. Used by sandbox mode.

Engines accept a `HaystackCorpus` per-call (not in constructor) — same engine instance serves benchmark and sandbox traffic without re-instantiation.

Output: new `mcp_servers/rag_pill/corpus/` package, all 5 engines refactored to consume it, both adapters with unit tests.
</objective>

```python
"""HaystackCorpus — the haystack seam.

A `HaystackCorpus` owns:
- the set of source documents the engines index and retrieve from
- the corpus version hash (for invalidating cached indices on change)
- whether ground-truth evaluation queries are available for this corpus

Engines accept a HaystackCorpus per-call so the same engine instance serves
both benchmark (fixed corpus) and sandbox (ephemeral corpus) traffic.
"""

from __future__ import annotations

import hashlib
from dataclasses import dataclass
from typing import Iterable, Protocol


@dataclass(frozen=True)
class CorpusDocument:
    """One source document inside a HaystackCorpus."""

    id: str          # stable id, e.g. "geography_fr.md" or "upload-{sha256}"
    text: str        # full document text
    meta: dict = None  # type: ignore[assignment]

    def __post_init__(self) -> None:
        if self.meta is None:
            object.__setattr__(self, "meta", {})


@dataclass(frozen=True)
class ExpectedSpan:
    """One [char_start, char_end) interval where an answer is known to live."""

    source_doc_id: str
    char_start: int
    char_end: int


@dataclass(frozen=True)
class EvaluationQuery:
    """A query whose ground-truth answer spans are known."""

    id: str
    query_text: str
    goal_text: str
    expected_spans: tuple[ExpectedSpan, ...]
    notes: str | None = None


class HaystackCorpus(Protocol):
    id: str
    has_ground_truth: bool

    def iter_documents(self) -> Iterable[CorpusDocument]:
        """Yield every document. Order MUST be stable across calls."""
        ...

    def get_document(self, doc_id: str) -> CorpusDocument | None:
        """Resolve a doc_id back to its text. None if unknown."""
        ...

    def list_evaluation_queries(self) -> list[EvaluationQuery]:
        """Empty when has_ground_truth is False."""
        ...

    @property
    def version_hash(self) -> str:
        """sha256 over sorted(doc_id + text) for cache invalidation."""
        ...


def compute_version_hash(documents: Iterable[CorpusDocument]) -> str:
    """Shared helper for adapters that don't override version_hash."""
    hasher = hashlib.sha256()
    for doc in sorted(documents, key=lambda d: d.id):
        hasher.update(doc.id.encode("utf-8"))
        hasher.update(b"\x00")
        hasher.update(doc.text.encode("utf-8"))
        hasher.update(b"\x01")
    return hasher.hexdigest()
```

```python
"""FixedCorpus — reads *.md from a directory and (optionally) queries.yaml."""

from __future__ import annotations

from functools import cached_property
from pathlib import Path

import yaml

from mcp_servers.rag_pill.corpus.base import (
    CorpusDocument,
    EvaluationQuery,
    ExpectedSpan,
    compute_version_hash,
)


class FixedCorpus:
    id = "fixed"
    has_ground_truth = True   # has_ground_truth flips to False if queries.yaml missing

    def __init__(self, root: Path) -> None:
        self._root = Path(root)
        if not self._root.exists():
            raise FileNotFoundError(f"FixedCorpus root does not exist: {self._root}")
        queries_path = self._root / "evaluation" / "queries.yaml"
        self._queries_path = queries_path if queries_path.exists() else None
        if self._queries_path is None:
            self.has_ground_truth = False

    @cached_property
    def _docs(self) -> list[CorpusDocument]:
        out: list[CorpusDocument] = []
        for path in sorted(self._root.glob("*.md")):
            out.append(CorpusDocument(id=path.name, text=path.read_text()))
        return out

    def iter_documents(self):
        return iter(self._docs)

    def get_document(self, doc_id: str) -> CorpusDocument | None:
        for d in self._docs:
            if d.id == doc_id:
                return d
        return None

    def list_evaluation_queries(self) -> list[EvaluationQuery]:
        if self._queries_path is None:
            return []
        data = yaml.safe_load(self._queries_path.read_text())
        if not isinstance(data, list):
            raise ValueError(f"{self._queries_path} must be a top-level list")
        out: list[EvaluationQuery] = []
        for entry in data:
            expected = tuple(
                ExpectedSpan(
                    source_doc_id=s["source_doc_id"],
                    char_start=int(s["char_start"]),
                    char_end=int(s["char_end"]),
                )
                for s in entry.get("expected_spans", [])
            )
            out.append(EvaluationQuery(
                id=entry["id"],
                query_text=entry["query_text"],
                goal_text=entry.get("goal_text", ""),
                expected_spans=expected,
                notes=entry.get("notes"),
            ))
        return out

    @cached_property
    def version_hash(self) -> str:
        return compute_version_hash(self._docs)
```

```python
"""EphemeralCorpus — wraps user-uploaded text as a single-doc corpus."""

from __future__ import annotations

import hashlib
from functools import cached_property

from mcp_servers.rag_pill.corpus.base import (
    CorpusDocument,
    EvaluationQuery,
    compute_version_hash,
)


class EphemeralCorpus:
    id = "ephemeral"
    has_ground_truth = False

    def __init__(self, document_content: str) -> None:
        if not document_content.strip():
            raise ValueError("EphemeralCorpus requires non-empty document_content")
        doc_id = "upload-" + hashlib.sha256(document_content.encode("utf-8")).hexdigest()[:16]
        self._doc = CorpusDocument(id=doc_id, text=document_content)

    def iter_documents(self):
        return iter([self._doc])

    def get_document(self, doc_id: str):
        return self._doc if doc_id == self._doc.id else None

    def list_evaluation_queries(self) -> list[EvaluationQuery]:
        return []

    @cached_property
    def version_hash(self) -> str:
        return compute_version_hash([self._doc])
```
